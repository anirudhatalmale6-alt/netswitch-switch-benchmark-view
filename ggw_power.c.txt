/* ggw_power - what the machine is actually drawing, in watts.
   AI2ORBIT Co. (c) 2026.

   WHY THIS IS A NATIVE PROGRAM AND NOT A WEB PAGE. No browser hands a
   page a voltage, a current or a watt. Not one. That is not a gap in
   any particular browser, it is deliberate - the figure identifies
   hardware too precisely. A program running on the machine itself has
   the files, so this reads them.

   WHERE A WATT ACTUALLY COMES FROM, in the order this tries them:

     1. THE PROCESSOR'S OWN ENERGY COUNTER (powercap, called RAPL on
        Intel and on recent AMD). A counter of microjoules that only
        ever climbs. It is NOT a watt - it is energy. Watts come from
        reading it twice and dividing the difference by the time
        between: joules per second. A single read tells you nothing
        at all, which is why this program always waits and reads
        again.

     2. THE SENSOR CHIPS (hwmon). powerN_input is microwatts and can
        be read once. currN_input and inN_input are milliamps and
        millivolts, and multiply to watts.

     3. THE BATTERY (power_supply). power_now is microwatts directly.
        Where it is absent, voltage_now times current_now gives it.
        This is the one that works on a phone.

   EVERY FIGURE SAYS WHERE IT CAME FROM. A watt read from the
   processor's own counter and a watt worked out from a battery are
   not the same measurement and must never be printed as though they
   were.

   NOTHING IS INVENTED. A source that is not present prints NOT
   AVAILABLE. There is no fallback to a typical value, no estimate
   from the chip name, and no figure derived from how busy the
   machine looks - all three would produce a number that reads like a
   measurement and is not one.

   THE ROOT IS AN ARGUMENT so the arithmetic can be tested against a
   tree of known values. A reader that can only run on real hardware
   is a reader nobody has tested.

   Build:  cc -O2 -o ggw_power ggw_power.c
   Run:    ./ggw_power [--ms 500] [--root /some/fake/sys] [--json]
*/

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <time.h>

#define MAXSRC 32
#define NAMELEN 256
#define DIRLEN 1024
/* The buffers are sized so no path can be silently cut short. A
   truncated path opens the wrong file or none, and the reader
   would report NOT AVAILABLE for a sensor that is really there. */
#define PATHLEN (DIRLEN + NAMELEN + 64)

typedef struct {
    char where[DIRLEN];      /* which file it came from */
    char kind[40];        /* processor counter, sensor chip, battery */
    char name[NAMELEN];        /* what the kernel calls it */
    double watts;         /* -1 when it could not be had */
    int derived;          /* 1 when multiplied out rather than read */
    char note[200];
} Src;

static Src srcs[MAXSRC];
static int nsrc = 0;
static char root[DIRLEN] = "";

static void joinp(char *out, const char *a, const char *b)
{
    if (root[0]) snprintf(out, PATHLEN, "%s%s/%s", root, a, b);
    else         snprintf(out, PATHLEN, "%s/%s", a, b);
}

/* MISSING IS REPORTED SEPARATELY FROM THE VALUE, never folded into
   it. This first returned -1 to mean "not there", which collides with
   a perfectly legitimate reading: a battery that is DISCHARGING
   reports a negative current, and on a phone that is the normal case.
   The sentinel swallowed it and the reader found no battery at all on
   exactly the device this is for. Returns 1 when a figure was had. */
static int readnum(const char *dir, const char *leaf, double *out)
{
    char p[PATHLEN]; double v; FILE *f;
    joinp(p, dir, leaf);
    f = fopen(p, "r");
    if (!f) return 0;
    if (fscanf(f, "%lf", &v) != 1) { fclose(f); return 0; }
    fclose(f);
    *out = v;
    return 1;
}

static int readstr(const char *dir, const char *leaf, char *out, size_t n)
{
    char p[PATHLEN]; FILE *f; size_t i;
    joinp(p, dir, leaf);
    f = fopen(p, "r");
    if (!f) return 0;
    if (!fgets(out, (int)n, f)) { fclose(f); out[0] = 0; return 0; }
    fclose(f);
    for (i = strlen(out); i > 0 && (out[i-1] == '\n' || out[i-1] == '\r'); i--) out[i-1] = 0;
    return 1;
}

static double now_s(void)
{
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return (double)t.tv_sec + (double)t.tv_nsec / 1e9;
}

static void add(const char *where, const char *kind, const char *name,
                double watts, int derived, const char *note)
{
    if (nsrc >= MAXSRC) return;
    snprintf(srcs[nsrc].where, sizeof srcs[nsrc].where, "%s", where);
    snprintf(srcs[nsrc].kind,  sizeof srcs[nsrc].kind,  "%s", kind);
    snprintf(srcs[nsrc].name,  sizeof srcs[nsrc].name,  "%s", name);
    snprintf(srcs[nsrc].note,  sizeof srcs[nsrc].note,  "%s", note ? note : "");
    srcs[nsrc].watts = watts;
    srcs[nsrc].derived = derived;
    nsrc++;
}

/* A PATH THAT WOULD NOT FIT IS REFUSED, NOT CUT SHORT. A truncated
   path names a different file, or none, and the reader would then
   report NOT AVAILABLE for a sensor that is really there - a silent
   wrong answer rather than a loud one. Returns 0 when it did not
   fit, and the caller skips that entry. */
static int mkpath(char *out, size_t cap, const char *fmt, const char *a)
{
    int n = snprintf(out, cap, fmt, a);
    return (n > 0 && (size_t)n < cap);
}

static int listdir(const char *rel, char names[][NAMELEN], int max)
{
    char p[PATHLEN]; DIR *d; struct dirent *e; int n = 0;
    if (root[0]) snprintf(p, sizeof p, "%s%s", root, rel);
    else         snprintf(p, sizeof p, "%s", rel);
    d = opendir(p);
    if (!d) return 0;
    while ((e = readdir(d)) && n < max) {
        if (e->d_name[0] == '.') continue;
        snprintf(names[n], NAMELEN, "%s", e->d_name);
        n++;
    }
    closedir(d);
    return n;
}

/* ---- 1. the processor's own energy counter ----
   TWO READS, ALWAYS. The file holds microjoules since boot. One read
   is a number with no meaning on its own. */
static void do_powercap(double gap_s)
{
    char names[MAXSRC][NAMELEN], dir[DIRLEN], label[NAMELEN], note[200];
    int n = listdir("/sys/class/powercap", names, MAXSRC), i;
    double e0[MAXSRC], t0, t1, e1, w;
    int kept[MAXSRC], nk = 0;

    for (i = 0; i < n; i++) {
        if (strncmp(names[i], "intel-rapl", 10) != 0 &&
            strncmp(names[i], "amd-rapl", 8) != 0) continue;
        if (!strchr(names[i], ':')) continue;         /* the zone, not the class */
        if (!mkpath(dir, sizeof dir, "/sys/class/powercap/%s", names[i])) continue;
        if (!readnum(dir, "energy_uj", &e0[nk])) continue;
        kept[nk] = i; nk++;
    }
    if (!nk) {
        add("/sys/class/powercap", "processor counter", "intel-rapl or amd-rapl",
            -1, 0, "no energy counter on this machine");
        return;
    }
    t0 = now_s();
    struct timespec ts;
    ts.tv_sec = (time_t)gap_s;
    ts.tv_nsec = (long)((gap_s - (double)ts.tv_sec) * 1e9);
    nanosleep(&ts, NULL);
    t1 = now_s();

    for (i = 0; i < nk; i++) {
        if (!mkpath(dir, sizeof dir, "/sys/class/powercap/%s", names[kept[i]])) continue;
        if (!readstr(dir, "name", label, sizeof label))
            if (!mkpath(label, sizeof label, "%s", names[kept[i]])) continue;
        if (!readnum(dir, "energy_uj", &e1)) {
            add(dir, "processor counter", label, -1, 0, "counter vanished between reads");
            continue;
        }
        if (e1 < e0[i]) {
            /* THE COUNTER WRAPS. Reporting the negative difference as
               a watt figure would print a machine generating power. */
            add(dir, "processor counter", label, -1, 0,
                "counter wrapped between the two reads, so this interval cannot be used");
            continue;
        }
        w = (e1 - e0[i]) / 1e6 / (t1 - t0);
        snprintf(note, sizeof note, "two reads %.3f s apart, %.0f microjoules used", t1 - t0, e1 - e0[i]);
        add(dir, "processor counter", label, w, 0, note);
    }
}

/* ---- 2. the sensor chips ---- */
static void do_hwmon(void)
{
    char names[MAXSRC][NAMELEN], dir[DIRLEN], label[NAMELEN], leaf[32], note[200];
    int n = listdir("/sys/class/hwmon", names, MAXSRC), i, k, found = 0;
    double uw, ma, mv;

    for (i = 0; i < n; i++) {
        if (!mkpath(dir, sizeof dir, "/sys/class/hwmon/%s", names[i])) continue;
        if (!readstr(dir, "name", label, sizeof label))
            if (!mkpath(label, sizeof label, "%s", names[i])) continue;
        for (k = 1; k <= 8; k++) {
            snprintf(leaf, sizeof leaf, "power%d_input", k);
            if (readnum(dir, leaf, &uw)) {
                snprintf(note, sizeof note, "%s read directly in microwatts", leaf);
                add(dir, "sensor chip", label, uw / 1e6, 0, note);
                found = 1;
            }
        }
        for (k = 1; k <= 8; k++) {
            snprintf(leaf, sizeof leaf, "curr%d_input", k);
            int gotA = readnum(dir, leaf, &ma);
            snprintf(leaf, sizeof leaf, "in%d_input", k);
            int gotV = readnum(dir, leaf, &mv);
            if (gotA && gotV) {
                snprintf(note, sizeof note, "%.0f mA times %.0f mV, multiplied out", ma, mv);
                add(dir, "sensor chip", label, ma / 1000.0 * mv / 1000.0, 1, note);
                found = 1;
            }
        }
    }
    if (!found)
        add("/sys/class/hwmon", "sensor chip", "any", -1, 0,
            "no power, current or voltage sensor on this machine");
}

/* ---- 3. the battery. This is the one that works on a phone. ---- */
static void do_battery(void)
{
    char names[MAXSRC][NAMELEN], dir[DIRLEN], type[64], note[200];
    int n = listdir("/sys/class/power_supply", names, MAXSRC), i, found = 0;
    double uw, uv, ua;

    for (i = 0; i < n; i++) {
        if (!mkpath(dir, sizeof dir, "/sys/class/power_supply/%s", names[i])) continue;
        if (!readstr(dir, "type", type, sizeof type)) type[0] = 0;
        if (readnum(dir, "power_now", &uw)) {
            snprintf(note, sizeof note, "power_now, read directly in microwatts, type %s",
                     type[0] ? type : "unknown");
            add(dir, "battery", names[i], uw / 1e6, 0, note);
            found = 1;
            continue;
        }
        if (readnum(dir, "voltage_now", &uv) && readnum(dir, "current_now", &ua)) {
            /* DISCHARGING IS REPORTED AS A NEGATIVE CURRENT. The
               machine is drawing that power either way, so the sign
               is dropped and the magnitude kept. */
            if (ua < 0) ua = -ua;
            snprintf(note, sizeof note, "%.3f V times %.3f A, multiplied out, type %s",
                     uv / 1e6, ua / 1e6, type[0] ? type : "unknown");
            add(dir, "battery", names[i], uv / 1e6 * (ua / 1e6), 1, note);
            found = 1;
        }
    }
    if (!found)
        add("/sys/class/power_supply", "battery", "any", -1, 0,
            "no battery or supply on this machine");
}

static void print_plain(void)
{
    int i, any = 0;
    printf("ggw_power - what this machine is drawing\n");
    printf("AI2Orbit\n\n");
    for (i = 0; i < nsrc; i++) {
        if (srcs[i].watts < 0) {
            printf("  %-18s %-22s NOT AVAILABLE\n", srcs[i].kind, srcs[i].name);
            printf("  %-18s %-22s %s\n\n", "", "", srcs[i].note);
        } else {
            any = 1;
            printf("  %-18s %-22s %10.3f W%s\n", srcs[i].kind, srcs[i].name,
                   srcs[i].watts, srcs[i].derived ? "   (multiplied out)" : "");
            printf("  %-18s %-22s %s\n", "", "", srcs[i].where);
            printf("  %-18s %-22s %s\n\n", "", "", srcs[i].note);
        }
    }
    if (!any)
        printf("  No watt figure could be had on this machine. Nothing has been\n"
               "  estimated in its place.\n");
}

static void print_json(void)
{
    int i;
    printf("{\"source\":\"ggw_power\",\"readings\":[");
    for (i = 0; i < nsrc; i++) {
        printf("%s{\"kind\":\"%s\",\"name\":\"%s\",\"where\":\"%s\",\"derived\":%d,",
               i ? "," : "", srcs[i].kind, srcs[i].name, srcs[i].where, srcs[i].derived);
        if (srcs[i].watts < 0) printf("\"watts\":null,");
        else                   printf("\"watts\":%.6f,", srcs[i].watts);
        printf("\"note\":\"%s\"}", srcs[i].note);
    }
    printf("]}\n");
}

int main(int argc, char **argv)
{
    double gap = 0.5;
    int json = 0, i;
    for (i = 1; i < argc; i++) {
        if (!strcmp(argv[i], "--ms") && i + 1 < argc) gap = atof(argv[++i]) / 1000.0;
        else if (!strcmp(argv[i], "--root") && i + 1 < argc)
            snprintf(root, sizeof root, "%s", argv[++i]);
        else if (!strcmp(argv[i], "--json")) json = 1;
        else if (!strcmp(argv[i], "--help")) {
            printf("ggw_power [--ms 500] [--root DIR] [--json]\n");
            return 0;
        }
    }
    if (gap < 0.05) gap = 0.05;
    do_powercap(gap);
    do_hwmon();
    do_battery();
    if (json) print_json(); else print_plain();
    return 0;
}
